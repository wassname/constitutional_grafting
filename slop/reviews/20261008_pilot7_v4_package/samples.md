```text

Do I judge an act the same way when I am the onlooker, the victim, or the one who acted? Where my principles conflict, which is the better reading? async function verifyToken(req, res, next) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) return res.status(401).json({ error: "Missing token" });
  try {

    req.user = decoded; // uid, email, etc.
    next();
  } catch {
    res.status(401).json({ error: "Invalid token" });
  }
}

async function requireRole(role) {
  return async (req, res, next) => {

    if (!snap.exists || snap.data().role !== role) {

// POST /reviews — one review per user per business
router.post("/", verifyToken, async (req, res) => {
  try {
    const { businessId, rating, text } = req.body;
    if (!businessId || !rating || rating < 1 || rating > 5) {
      return res.status(400).json({ error: "businessId and rating 1–5 required" });
    }

    const bizRef = db.collection("businesses").doc(businessId);
    const reviewId = `${businessId}_${req.user.uid}`;
    const reviewRef = db.collection("reviews").doc(reviewId);

    // Deterministic ID prevents duplicate reviews via the transaction itself
    await db.runTransaction(async (tx) => {
      const [bizDoc, existing] = await Promise.all([tx.get(bizRef), tx.get(reviewRef)]);
      if (!bizDoc.exists) throw new Error("Business not found");
      if (existing.exists) throw new Error("Already reviewed");

      const old = bizDoc.data();
      const newCount = old.reviewCount + 1;
      const newAvg = (old.avgRating * old.reviewCount + rating) / newCount;

      tx.set(reviewRef, {
        businessId, userId: req.user.uid, rating,
        text: text || "", createdAt: FieldValue.serverTimestamp(),
      });
      tx.update(bizRef, {
        reviewCount: newCount,
        avgRating: Math.round(newAvg * 10) / 10,
      });
    });

    res.status(201).json({ ok: true });
  } catch (e) {
    res.status(400).json({ error: e.message });
  }
});

import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class AuthService {
  final _auth = FirebaseAuth.instance;
  final _storage = const FlutterSecureStorage();

  Future<void> register(String email, String password, String name) async {
    final cred = await _auth.createUserWithEmailAndPassword(email: email, password: password);
    await cred.user?.updateDisplayName(name);
    await _saveToken();
  }

  Future<void> login(String email, String password) async {
    await _auth.signInWithEmailAndPassword(email: email, password: password);
    await _saveToken();
  }

  Future<void> _saveToken() async {
    final token = await _auth.currentUser?.getIdToken();
    if (token != null) await _storage.write(key: "id_token", value: token);
  }

  Future<String?> getToken() async {
    // Refresh if older than ~50 min
    final user = _auth.currentUser;
    if (user == null) return null;
    final token = await user.getIdToken(true);
    await _storage.write(key: "id_token", value: token);
    return token;
  }

  Future<void> logout() async {
    await _storage.delete(key: "id_token");
    await _auth.signOut();
  }
}

      if (category != null) "category": category,
      if (lat != null) "lat": "$lat",
      if (lng != null) "lng": "$lng",
      if (radiusKm != null) "radiusKm": "$radiusKm",

```js
function haversineKm(lat1, lng1, lat2, lng2) {
  const R = 6371;
  const toRad = (x) => (x * Math.PI) / 180;
  const dLat = toRad(lat2 - lat1);
  const dLng = toRad(lng2 - lng1);
  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) ** 2;
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

const { auth, db } = require("../firebase");

async function verifyToken(req, res, next) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) return res.status(401).json({ error: "Missing token" });
  try {
    const decoded = await auth.verifyIdToken(token);
    req.user = decoded; // uid, email, etc.
    next();
  } catch {
    res.status(401).json({ error: "Invalid token" });
  }
}

async function requireRole(role) {
  return async (req, res, next) => {
    const snap = await db.collection("users").doc(req.user.uid).get();
    if (!snap.exists || snap.data().role !== role) {
      return res.status(403).json({ error: "Forbidden" });
    }
    next();
  };
}

      .orderBy("avgRating", "desc")

### Step 3.7: `src/routes/reviews.js` — rate, review, recalculate averages

```js
const express = require("express");
const router = express.Router();
const { db, FieldValue } = require("../firebase");
const { verifyToken } = require("../middleware/auth");

// POST /reviews — one review per user per business
router.post("/", verifyToken, async (req, res) => {
  try {
    const { businessId, rating, text } = req.body;
    if (!businessId || !rating || rating < 1 || rating > 5) {
      return res.status(400).json({ error: "businessId and rating 1–5 required" });
    }

    const bizRef = db.collection("businesses").doc(businessId);
    const reviewId = `${businessId}_${req.user.uid}`;
    const reviewRef = db.collection("reviews").doc(reviewId);

    const userSnap = await db.collection("users").doc(req.user.uid).get();
    const userName = userSnap.exists ? userSnap.data().displayName || "Anonymous" : "Anonymous";

    // Deterministic ID + transaction prevents duplicate reviews
    await db.runTransaction(async (tx) => {
      const [bizDoc, existing] = await Promise.all([tx.get(bizRef), tx.get(reviewRef)]);
      if (!bizDoc.exists) throw new Error("Business not found");
      if (existing.exists) throw new Error("Already reviewed");

      const old = bizDoc.data();
      const newCount = old.reviewCount + 1;
      const newAvg = (old.avgRating * old.reviewCount + rating) / newCount;

      tx.set(reviewRef, {
        businessId, userId: req.user.uid, userName, rating,
        text: text || "", createdAt: FieldValue.serverTimestamp(),
      });
      tx.update(bizRef, {
        reviewCount: newCount,
        avgRating: Math.round(newAvg * 10) / 10,
      });
    });

    res.status(201).json({ ok: true });
  } catch (e) {
    res.status(400).json({ error: e.message });
  }
});

import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class AuthService {
  final _auth = FirebaseAuth.instance;
  final _storage = const FlutterSecureStorage();

  Future<void> register(String email, String password, String name) async {
    final cred = await _auth.createUserWithEmailAndPassword(email: email, password: password);
    await cred.user?.updateDisplayName(name);
    await _saveToken();
  }

  Future<void> login(String email, String password) async {
    await _auth.signInWithEmailAndPassword(email: email, password: password);
    await _saveToken();
  }

  Future<void> _saveToken() async {
    final token = await _auth.currentUser?.getIdToken();
    if (token != null) await _storage.write(key: "id_token", value: token);
  }

  Future<String?> getToken() async {
    final user = _auth.currentUser;
    if (user == null) return null;
    final token = await user.getIdToken(true); // forces refresh
    await _storage.write(key: "id_token", value: token);
    return token;
  }

  Future<void> logout() async {
    await _storage.delete(key: "id_token");
    await _auth.signOut();
  }
}

import 'dart:convert';
import 'package:http/http.dart' as http;
import 'auth_service.dart';

class ApiService {
  static const base = apiBaseUrl;
  final _auth = AuthService();

  Future<Map<String, String>> _headers() async {
    final token = await _auth.getToken();
    return {
      "Content-Type": "application/json",
      if (token != null) "Authorization": "Bearer $token",
    };
  }

  Future<dynamic> get(String path, {Map<String, String>? query}) async {
    var uri = Uri.parse("$base$path");
    if (query != null && query.isNotEmpty) uri = uri.replace(queryParameters: query);
    final res = await http.get(uri, headers: await _headers());
    _check(res);
    return jsonDecode(res.body);
  }

  Future<dynamic> post(String path, Map<String, dynamic> body) async {
    final res = await http.post(Uri.parse("$base$path"),
        headers: await _headers(), body: jsonEncode(body));
    _check(res);
    return res.body.isEmpty ? null : jsonDecode(res.body);
  }

  Future<dynamic> put(String path, Map<String, dynamic> body) async {
    final res = await http.put(Uri.parse("$base$path"),
        headers: await _headers(), body: jsonEncode(body));
    _check(res);
    return res.body.isEmpty ? null : jsonDecode(res.body);
  }

  Future<dynamic> delete(String path) async {
    final res = await http.delete(Uri.parse("$base$path"), headers: await _headers());
    _check(res);
    return res.body.isEmpty ? null : jsonDecode(res.body);
  }

  void _check(http.Response res) {
    if (res.statusCode >= 400) {
      throw Exception("API error ${res.statusCode}: ${res.body}");
    }
  }

  // Domain methods
  Future<List<dynamic>> searchBusinesses({String? q, String? category, double? lat, double? lng, double? radiusKm}) async {
    final query = {
      if (q != null && q.isNotEmpty) "q": q,
      if (category != null) "category": category,
      if (lat != null) "lat": "$lat",
      if (lng != null) "lng": "$lng",
      if (radiusKm != null) "radiusKm": "$radiusKm",
    };
    final data = await get("/businesses", query: query);
    return data["businesses"];
  }

  Future<List<dynamic>> myBusinesses() async {
    final data = await get("/businesses/mine");
    return data["businesses"];
  }

  Future<void> bookmark(String businessId) => post("/users/bookmarks", {"businessId": businessId});
  Future<void> unbookmark(String businessId) => delete("/users/bookmarks/$businessId");
  Future<void> review(String businessId, int rating, String text) =>
      post("/reviews", {"businessId": businessId, "rating": rating, "text": text});
  Future<void> claim(String businessId) => post("/businesses/$businessId/claim", {});
}

Gravitational Energy Storage (GES) stores energy by lifting a mass and recovers it as the mass descends, driving a generator. Key systems in the literature:

| System | Principle | Typical scale | Notes |
|---|---|---|---|

## 4. Methodology / Control Strategy

### 4.1 System description

A motor drives a drum (radius r, inertia J_d) through a gearbox (ratio N). A cable lifts mass m. States: drum angle θ, speed ω = θ̇.

### 4.2 Equation of motion

**Mechanical side** (Newton's second law, referred to the motor shaft):

Motor torque T_m, load torque from gravity referred through gearbox: T_g = m g r / N.

Friction/damping: viscous b, plus Coulomb friction T_c (often neglected in linear model).

```
(J_m + J_d + m r²/N²) θ̈ + b θ̇ = T_m − (m g r / N)
```

Let **J_eq = J_m + J_d + m r²/N²**, **T_g = m g r / N**:

> **J_eq θ̈ + b θ̇ = T_m − T_g**

**Electrical side** (DC motor):

```
L di/dt + R i = V − K_e θ̇
T_m = K_t i
```

### 4.3 Transfer function

Taking Laplace transforms, with armature current eliminated:

**Motor + load transfer function (V → θ):**

```
θ(s)      K_t
──── = ─────────────────────────────────────
V(s)   s [ (R + Ls)(J_eq s + b) + K_t K_e ]
```

For small L (current-mode drive, i.e. torque control), the dominant plant reduces to:

```
θ(s)      1/K_e
──── ≈ ─────────────────────────
V(s)   s (τ s + 1),   τ = J_eq R/(K_t K_e)
```

**Speed transfer function (T_m → ω):**

```
ω(s)      1
───── = ─────────
T_m(s)   J_eq s + b
```

The gravity term T_g is a constant disturbance entering at the torque summing junction — this is the key control challenge: on lifting, gravity opposes the motor; on lowering, gravity assists (regenerative braking / generator mode).

### 4.4 Control strategy

- **Cascade control:** inner current (torque) loop, outer speed loop, outermost position loop for precise parking of the mass.
- **Disturbance feedforward:** add T_g = m g r / N to the commanded torque, with sign depending on direction — cancels gravity instantly instead of waiting for integral action.
- **Charge mode:** speed reference ω_ref > 0 (lift); **Discharge mode:** motor acts as generator, controlled descent with torque limit; **Hold mode:** zero speed with holding torque or mechanical brake.
- **PID gains:** tuned via Ziegler–Nichols or pole placement; verify overshoot < 5%, settling time < 1 s (lab scale).

1. Build plant TF in MATLAB.
2. Design PID (speed loop) + P (position loop).
3. Simulate charge → hold → discharge cycle; plot θ(t), ω(t), i(t).
4. Compute stored/recovered energy: E = ∫ T_m ω dt.

r   = 0.2;    N   = 10;       % drum radius (m), gearbox ratio

Kp_speed = 50; Ki_speed = 200;
C_speed = pid(Kp_speed, Ki_speed);

% Position loop (P) around closed speed loop
Kp_pos = 8;
C_pos = Kp_pos;

% Closed speed loop
T_speed = feedback(C_speed*Gs, 1);
T_pos   = feedback(C_pos*T_speed/s, 1);

step(T_pos, 5); title('Position step response (charge cycle)');
grid on;

%% Simulink-style time simulation (ode45 equivalent via lsim)
t = 0:0.001:20;
% Reference: lift 10 rad in 0-5s, hold 5-10s, lower 10-15s, hold
wref = zeros(size(t));

wref(t>=10 & t<15) = -2;  % discharge (controlled descent)

% Plant with gravity disturbance (sign flips with direction)
sys_cl = feedback(C_speed*Gs, 1);
[resp] = lsim(sys_cl, wref, t);

% Gravity disturbance as step torque input, filtered by closed loop
dist = Tg*sign(wref);         % opposes motion
dist_resp = lsim( feedback(Gs, C_speed), dist, t );
w = resp - dist_resp;

figure; plot(t, wref, 'k--', t, w, 'b'); grid on;
xlabel('t (s)'); ylabel('\omega (rad/s)');
legend('\omega_{ref}','\omega (with gravity disturbance)');
title('Speed tracking: charge / hold / discharge');

%% Energy accounting
theta = cumtrapz(t, w);
E_stored = trapz(t, Tg .* w .* (w>0));   % mechanical energy in
E_out    = trapz(t, Tg .* (-w) .* (w<0));% mechanical energy out
